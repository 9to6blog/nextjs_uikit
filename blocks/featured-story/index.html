블록 컴포넌트 소스
blocks/featured-story.tsx
import { Badge } from "../badge.js";
import { BlockLink, type Article } from "./shared.js";
export type FeaturedStoryProps = {
  article: Article;
  eyebrow?: string;
  className?: string;
};
export function FeaturedStory({
  article,
  eyebrow = "추천 글",
  className,
}: FeaturedStoryProps) {
  return (
    <article className={`n-block n-block-featured ${className ?? ""}`}>
      <div className="n-block-featured-art">{article.artwork}</div>
      <div>
        <Badge>{eyebrow}</Badge>
        <h2>{article.title}</h2>
        <p>{article.excerpt}</p>
        <div className="n-block-meta">
          <span>{article.category}</span>
          <span>{article.readingTime}</span>
        </div>
        <BlockLink href={article.href}>이야기 읽기</BlockLink>
      </div>
    </article>
  );
}

예제에 있는 block-demo-* 클래스는 문서 안내용입니다. 블록의 스타일은 blocks.css에 포함됩니다. 소스를 직접 복사할 때는 같은 폴더의 shared 모듈도 함께 가져오세요.
